Func<UnityEngine.Object, object> identity = delegate(UnityEngine.Object obj)
{
    if (obj == null) return null;
    var gid = GlobalObjectId.GetGlobalObjectIdSlow(obj);
    return new { instanceId = obj.GetInstanceID(), type = obj.GetType().FullName,
        globalObjectId = gid.ToString(), assetGuid = gid.assetGUID.ToString(),
        localFileId = gid.targetObjectId.ToString() };
};

Func<object> snapshot = delegate()
{
    var scenes = new List<object>();
    var objects = new List<GameObject>();
    for (int i = 0; i < UnityEngine.SceneManagement.SceneManager.sceneCount; i++)
    {
        var scene = UnityEngine.SceneManagement.SceneManager.GetSceneAt(i);
        var roots = scene.isLoaded ? scene.GetRootGameObjects() : new GameObject[0];
        scenes.Add(new { path = scene.path, guid = AssetDatabase.AssetPathToGUID(scene.path),
            loaded = scene.isLoaded, dirty = scene.isDirty, handle = scene.handle,
            active = scene == UnityEngine.SceneManagement.SceneManager.GetActiveScene(),
            rootIds = roots.Select(x => x.GetInstanceID()).OrderBy(x => x).ToArray() });
        foreach (var root in roots)
            foreach (var tr in root.GetComponentsInChildren<Transform>(true)) objects.Add(tr.gameObject);
    }
    var managers = new List<object>();
    foreach (var go in objects.OrderBy(x => x.GetInstanceID()))
    {
        foreach (var manager in go.GetComponents<MoreMountains.Tools.MMSoundManager>())
        {
            using (var so = new SerializedObject(manager))
            {
                var scriptProperty = so.FindProperty("m_Script");
                var settingsProperty = so.FindProperty("settingsSo");
                var unparentProperty = so.FindProperty("AutomaticallyUnparentOnAwake");
                var sizeProperty = so.FindProperty("AudioSourcePoolSize");
                var expandProperty = so.FindProperty("PoolCanExpand");
                var enabledProperty = so.FindProperty("m_Enabled");
                var transform = go.transform;
                managers.Add(new { scenePath = go.scene.path,
                    sceneGuid = AssetDatabase.AssetPathToGUID(go.scene.path),
                    gameObject = identity(go), component = identity(manager), transform = identity(transform),
                    name = go.name, activeSelf = go.activeSelf, activeInHierarchy = go.activeInHierarchy,
                    enabled = manager.enabled,
                    serializedEnabled = enabledProperty == null ? (bool?)null : enabledProperty.boolValue,
                    script = scriptProperty == null ? null : identity(scriptProperty.objectReferenceValue),
                    settingsSo = settingsProperty == null ? null : identity(settingsProperty.objectReferenceValue),
                    automaticallyUnparentOnAwake = unparentProperty == null ? (bool?)null : unparentProperty.boolValue,
                    audioSourcePoolSize = sizeProperty == null ? (int?)null : sizeProperty.intValue,
                    poolCanExpand = expandProperty == null ? (bool?)null : expandProperty.boolValue,
                    parentId = transform.parent == null ? (int?)null : transform.parent.gameObject.GetInstanceID(),
                    childIds = Enumerable.Range(0, transform.childCount).Select(n => transform.GetChild(n).gameObject.GetInstanceID()).ToArray(),
                    localPosition = new float[] { transform.localPosition.x, transform.localPosition.y, transform.localPosition.z },
                    localRotation = new float[] { transform.localRotation.x, transform.localRotation.y, transform.localRotation.z, transform.localRotation.w },
                    localScale = new float[] { transform.localScale.x, transform.localScale.y, transform.localScale.z },
                    missingScripts = GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(go),
                    componentIds = go.GetComponents<Component>().Select(c => c == null ? 0 : c.GetInstanceID()).ToArray() });
            }
        }
    }
    var current = MoreMountains.Tools.MMSoundManager.Current;
    return new { sceneSetup = UnityEditor.SceneManagement.EditorSceneManager.GetSceneManagerSetup()
            .Select(s => new { path = s.path, loaded = s.isLoaded, active = s.isActive }).ToArray(),
        scenes = scenes.ToArray(),
        gameObjectIds = objects.Select(g => g.GetInstanceID()).OrderBy(x => x).ToArray(),
        managers = managers.ToArray(), currentInstanceId = current == null ? (int?)null : current.GetInstanceID() };
};
var before = snapshot();
var after = snapshot();
return new { status = "READ_ONLY_COMPLETE", before = before, after = after,
    snapshotEqual = Newtonsoft.Json.Linq.JToken.DeepEquals(Newtonsoft.Json.Linq.JToken.FromObject(before), Newtonsoft.Json.Linq.JToken.FromObject(after)) };
