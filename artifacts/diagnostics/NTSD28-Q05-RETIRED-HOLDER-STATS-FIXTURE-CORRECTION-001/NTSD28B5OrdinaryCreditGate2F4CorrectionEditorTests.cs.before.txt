        public void CharacterRecovery_PpThresholdBoundary(
            string path,
            int exactGate,
            int initialPp,
            int expectedPp)
        {
            var world = new SimulationWorld();
            world.ConfigureBattleEcsCharacterRecoveryPassForDiagnostics(
                path == "Legacy"
                    ? BattleEcsCharacterRecoveryPassMode.Legacy
                    : BattleEcsCharacterRecoveryPassMode.DataOriented);
            LF2Character entity = path == "Derived"
                ? new DerivedRecoveryCharacter()
                : new LF2Character();
            InitializeCharacter(entity, 9850, 0);
            world.Register(entity);
            try
            {
                entity.Health.HP = 400;
                entity.Health.HPBound = 400;
                entity.Health.PP = initialPp;
                entity.HitStun = 0;
                entity.KillCount = exactGate == -1 ? 7 : -1;
                entity.Runtime.OrdinaryCreditGate2F4 = exactGate;

                world.LateEntityUpdateAll(3);

                BattleEcsCharacterRecoveryPassDiagnostics diagnostics =
                    world.BattleEcsCharacterRecoveryPassDiagnosticsForDiagnostics;
                Assert.That(diagnostics.ExactCharacterCount,
                    Is.EqualTo(path == "Ecs" ? 1 : 0),
                    $"recovery route={path}");
                Assert.That(diagnostics.CompatibilityFallbackCount,
                    Is.EqualTo(path == "Ecs" ? 0 : 1),
                    $"recovery route={path}");
                Assert.That(entity.Health.PP, Is.EqualTo(expectedPp),
                    $"PP threshold boundary: path={path}, gate={exactGate}, initialPP={initialPp}, expectedPP={expectedPp}, actualPP={entity.Health.PP}");
                Assert.That(entity.Health.HP, Is.EqualTo(400));
                Assert.That(entity.Health.HPBound, Is.EqualTo(400));
            }
            finally
            {
                world.Unregister(entity);
            }
        }

            string cpoint = Source(
                "Assets/NTSD/Scripts/Simulation/Ecs/Writers/BattleCpointWriter.cs");

                "Assets/NTSD/Scripts/Simulation/Ecs/Hit/BattleEcsHitExecutionPlan.cs");

                "Assets/NTSD/Scripts/Animation/LF2Objects/LF2Entity.cs");
            string recovery = Source(
                "Assets/NTSD/Scripts/Simulation/Ecs/Passes/BattleEcsCharacterRecoveryPass.cs");

                "Assets/NTSD/Scripts/Simulation/Runtime/BattleLogicEntityFactory.cs");

            Assert.That(Count(cpoint, "victim.KillCount == -1"), Is.Zero);

                "entity.KillCount != -1 &&",

            StringAssert.DoesNotContain("parent.KillCount", logicFactory);

            Assert.That(
                Count(damage, "victim.Runtime.OrdinaryCreditGate2F4 == -1"),
                Is.EqualTo(5));
            Assert.That(
                Count(hitPlan, "target.Runtime.OrdinaryCreditGate2F4 == -1"),
                Is.EqualTo(4));
            Assert.That(
                Count(hitPlan, "target?.Runtime?.OrdinaryCreditGate2F4 == -1"),
                Is.EqualTo(1));

                "Runtime.OrdinaryCreditGate2F4 != -1",

            StringAssert.Contains(
                "entity.Runtime.OrdinaryCreditGate2F4 != -1",
                recovery);
            StringAssert.Contains(

        private static void RunStandardDamageCase(
            int legacyKillCount,
            int exactGate,
            int expectedHolderCombo,
            int expectedHolderKills)
        {
            CreateDamageScope(
                legacyKillCount,
                exactGate,
                out SimulationWorld world,
                out GateCharacter attacker,
                out GateCharacter victim,
                out GateCharacter holder);

            bool applied = world.DamageWriter.ApplyStandardCharacterDamage(
                world,
                attacker,
                victim,
                victim.HitCounters,
                DamageInteraction());

            Assert.That(applied, Is.True);
            Assert.That(holder.ComboCountAtk, Is.EqualTo(expectedHolderCombo));
            Assert.That(holder.KillStat, Is.EqualTo(expectedHolderKills));
            Assert.That(victim.Runtime.OrdinaryCreditGate2F4, Is.EqualTo(exactGate));
            Assert.That(victim.KillCount, Is.EqualTo(legacyKillCount));
        }

        private static void RunReducedDamageCase(
            int legacyKillCount,
            int exactGate,
            int expectedHolderCombo,
            int expectedHolderKills)
        {
            CreateDamageScope(
                legacyKillCount,
                exactGate,
                out SimulationWorld world,
                out GateCharacter attacker,
                out GateCharacter victim,
                out GateCharacter holder);
            victim.Health.HP = 1;

            world.DamageWriter.ApplyAlternateDamage(
                world,
                attacker,
                victim,
                victim.HitCounters,
                DamageInteraction());

            Assert.That(holder.ComboCountAtk, Is.EqualTo(expectedHolderCombo));
            Assert.That(holder.KillStat, Is.EqualTo(expectedHolderKills));
            Assert.That(victim.Runtime.OrdinaryCreditGate2F4, Is.EqualTo(exactGate));
            Assert.That(victim.KillCount, Is.EqualTo(legacyKillCount));
        }

        private static void RunCpointDamageCase(
            int legacyKillCount,
            int exactGate,
            int expectedHolderKills)
        {
            CreateDamageScope(
                legacyKillCount,
                exactGate,
                out SimulationWorld world,
                out GateCharacter attacker,
                out GateCharacter victim,
                out GateCharacter holder);
            MethodInfo apply = typeof(BattleCpointWriter).GetMethod(
                "ApplyHeldInjury",
                BindingFlags.Instance | BindingFlags.NonPublic);
            Assert.That(apply, Is.Not.Null);

            apply.Invoke(
                new BattleCpointWriter(),
                new object[] { world, attacker, victim, 10, 0 });

            Assert.That(
                attacker.Runtime.KnockoutCount358,
                Is.EqualTo(expectedHolderKills));
            Assert.That(attacker.Runtime.InputScoreTotal348, Is.EqualTo(10));
            Assert.That(holder.KillStat, Is.Zero);
            Assert.That(holder.ComboCountAtk, Is.Zero);
            Assert.That(victim.Runtime.OrdinaryCreditGate2F4, Is.EqualTo(exactGate));
            Assert.That(victim.KillCount, Is.EqualTo(legacyKillCount));
        }

        private static void AssertRecoveryCase(
            bool useDerivedLegacyPath,
            int legacyKillCount,
            int exactGate,
            bool expectIncrease)
        {
            var world = new SimulationWorld();
            LF2Character entity = useDerivedLegacyPath
                ? new DerivedRecoveryCharacter()
                : new LF2Character();
            InitializeCharacter(entity, 9850, 0);
            world.Register(entity);
            entity.Health.HP = 400;
            entity.Health.HPBound = 500;
            entity.Health.PP = 200;
            entity.HitStun = 0;
            entity.KillCount = legacyKillCount;
            entity.Runtime.OrdinaryCreditGate2F4 = exactGate;

            world.LateEntityUpdateAll(12);

            Assert.That(
                entity.Health.PP,
                expectIncrease ? Is.GreaterThan(200) : Is.EqualTo(200),
                $"legacyPath={useDerivedLegacyPath}, kill={legacyKillCount}, gate={exactGate}");
        }

        private static LF2Entity SpawnLogicChild(
            int childOid,
            LF2ObjectType childType,
            int parentGate,
            int parentKillCount)
        {
            SimulationWorld world = CreateOpointWorld(
                Child(childOid, childType));
            ProbeOther parent = CreateParent(
                childOid,
                FrameWithOpoint(childOid),
                parentGate,
                parentKillCount);
            RegisterParent(world, parent);

            world.LateEntityUpdateAll(1);

            LF2Entity child = world.FindEntityByRuntimeSlotForQuery(51);
            Assert.That(child, Is.Not.Null);
            Assert.That(child.OwnerEntityIndex, Is.EqualTo(7));
            return child;
        }

        private static void AssertTwoHopLogicPropagation()
        {
            const int firstOid = 9904;
            const int secondOid = 9905;
            LF2FrameData firstFrame = FrameWithOpoint(secondOid);
            SimulationWorld world = CreateOpointWorld(
                Child(firstOid, LF2ObjectType.Character, firstFrame),
                Child(secondOid, LF2ObjectType.Character));
            ProbeOther root = CreateParent(
                firstOid,
                FrameWithOpoint(firstOid),
                parentGate: -1,
                parentKillCount: 7);
            RegisterParent(world, root);

            world.LateEntityUpdateAll(2);
            LF2Entity first = world.FindEntityByRuntimeSlotForQuery(51);
            Assert.That(first, Is.Not.Null);
            Assert.That(first.Runtime.OrdinaryCreditGate2F4, Is.EqualTo(ParentSlot));
            first.AttackingCounter = 0;
            first.FrameDelay = 0;
            first.Frame.D = firstFrame;
            first.Frame.N = 0;
            first.Runtime.Frame = 0;
            world.StructuralWriter.ProcessLateOpointSegment(
                world.LogicObjectPointRuntime,
                first,
                3);

            LF2Entity second = world.FindEntityByRuntimeSlotForQuery(52);
            Assert.That(second, Is.Not.Null);
            Assert.That(second.Runtime.OrdinaryCreditGate2F4,
                Is.EqualTo(ParentSlot));
            Assert.That(first.KillCount, Is.EqualTo(-1));
            Assert.That(second.KillCount, Is.EqualTo(-1));
        }

        private static void InitializeCharacter(
            LF2Character entity,
            int oid,
            int slot)
        {
            var frames = new List<LF2FrameData>(241);
            for (int id = 0; id <= 240; id++)
            {
                frames.Add(new LF2FrameData
                {
                    frameId = id,
                    state = LF2States.Standing,
                    wait = 100,
                    next = id,
                    pic = 999,
                });
            }
            entity.Name = $"Gate2F4_{oid}";
            entity.ObjectId = oid;
            entity.ModuleInitialize();
            entity.SetRequiredRuntimeSlot(slot);
            entity.FrameCache.Load(new LF2CharacterDataWrapper(
                oid,
                new LF2CharacterData
                {
                    name = entity.Name,
                    type_sub = 0,
                    frames = frames,
                }));
            entity.ImmediateFrame(0);
            entity.Initialize(500, 500);
            entity.Runtime.SetPosition(slot * 20, 0, 0);
            entity.Runtime.SyncIntegerPosition();
        }

        private static ChildDefinition Child(
            int oid,
            LF2ObjectType type,
            LF2FrameData frame = null)
        {
            return new ChildDefinition(
                oid,
                type,
                frame ?? FrameWithoutOpoint());
        }

        private static SimulationWorld CreateOpointWorld(
            params ChildDefinition[] children)
        {
            var wrappers = new Dictionary<int, LF2CharacterDataWrapper>();
            var definitions = new List<ObjectDefinition>(children.Length);
            for (int index = 0; index < children.Length; index++)
            {
                ChildDefinition child = children[index];
                var data = new LF2CharacterData
                {
                    name = $"Gate2F4Child{child.Oid}",
                    type_sub = (int)child.Type,
                    frames = new List<LF2FrameData> { child.Frame },
                };
                wrappers.Add(
                    child.Oid,
                    new LF2CharacterDataWrapper(child.Oid, data));
                definitions.Add(new ObjectDefinition(
                    child.Oid,
                    (int)child.Type,
                    "gate-2f4-opoint.dat"));
            }

            var resolver = new RuntimeCharacterConfigResolver(
                oid => wrappers.TryGetValue(
                    oid,
                    out LF2CharacterDataWrapper wrapper)
                    ? wrapper
                    : null);
            var world = new SimulationWorld(resolver);
            world.SetLogicOnlyEntityMaterialization(true);
            world.PrepareRuntimeDataCatalogForBattle(
                definitions,
                oid => wrappers.TryGetValue(
                    oid,
                    out LF2CharacterDataWrapper wrapper)
                    ? wrapper
                    : null);
            return world;
        }

        private static string Source(string relativePath)
        {
            string root = Directory.GetParent(Application.dataPath)?.FullName;
            return File.ReadAllText(Path.GetFullPath(
                Path.Combine(root ?? string.Empty, relativePath)));
        }

        private static int Count(string source, string value)
        {
            int count = 0;
            int offset = 0;
            while ((offset = source.IndexOf(
                       value,
                       offset,
                       StringComparison.Ordinal)) >= 0)
            {
                count++;
                offset += value.Length;
            }
            return count;
        }

    [InitializeOnLoad]
    internal sealed class NTSD28B5OrdinaryCreditGate2F4RequestRunner : ICallbacks
    {
        private const string RequestRelativePath =
            "Temp/NTSD28-B5-OrdinaryCreditGate2F4-v1.request";
        private const string ResultRelativePath =
            "Temp/NTSD28-B5-OrdinaryCreditGate2F4-v1.result";
        private const string FocusedTestClass =
            "NTSD.Test.Editor.NTSD28B5OrdinaryCreditGate2F4CorrectionEditorTests";

        private static readonly string RequestPath = ProjectPath(RequestRelativePath);
        private static readonly string ResultPath = ProjectPath(ResultRelativePath);
        private static readonly StringBuilder FailureDetails = new StringBuilder(4096);
        private static NTSD28B5OrdinaryCreditGate2F4RequestRunner activeCallbacks;
        private static TestRunnerApi activeApi;

        static NTSD28B5OrdinaryCreditGate2F4RequestRunner()
        {
            EditorApplication.update += PollRequest;
        }

        private static void PollRequest()
        {
            if (activeCallbacks != null || EditorApplication.isCompiling ||
                EditorApplication.isUpdating ||
                EditorApplication.isPlayingOrWillChangePlaymode ||
                !File.Exists(RequestPath))
            {
                return;
            }
            if (File.Exists(ResultPath))
                File.Delete(ResultPath);
            File.Delete(RequestPath);
            activeCallbacks = new NTSD28B5OrdinaryCreditGate2F4RequestRunner();
            activeApi = ScriptableObject.CreateInstance<TestRunnerApi>();
            activeApi.RegisterCallbacks(activeCallbacks);
            activeApi.Execute(new ExecutionSettings(new Filter
            {
                testMode = TestMode.EditMode,
                testNames = new[] { FocusedTestClass },
            })
            {
                runSynchronously = false,
            });
        }

        public void RunStarted(ITestAdaptor testsToRun)
        {
            FailureDetails.Clear();
        }

        public void RunFinished(ITestResultAdaptor result)
        {
            string text =
                $"state={result.ResultState}\n" +
                $"passed={result.PassCount}\n" +
                $"failed={result.FailCount}\n" +
                $"skipped={result.SkipCount}\n" +
                $"inconclusive={result.InconclusiveCount}\n" +
                $"message={result.Message}\n" +
                FailureDetails;
            File.WriteAllText(ResultPath, text, new UTF8Encoding(false));
            activeApi.UnregisterCallbacks(this);
            UnityEngine.Object.DestroyImmediate(activeApi);
            activeApi = null;
            activeCallbacks = null;
        }

        public void TestStarted(ITestAdaptor test)
        {
        }

        public void TestFinished(ITestResultAdaptor result)
        {
            if (result?.Test == null || result.Test.IsSuite || result.FailCount <= 0)
                return;
            FailureDetails.Append("--- failure ---\n");
            FailureDetails.Append("test=").Append(result.FullName).Append('\n');
            FailureDetails.Append("state=").Append(result.ResultState).Append('\n');
            FailureDetails.Append("message=").Append(result.Message).Append('\n');
            FailureDetails.Append("stack=").Append(result.StackTrace).Append('\n');
        }

        private static string ProjectPath(string relativePath)
        {
            string root = Directory.GetParent(Application.dataPath)?.FullName;
            return Path.GetFullPath(Path.Combine(root ?? string.Empty, relativePath));
        }
    }

        private static bool running;

        static NTSD28B5OrdinaryCreditGate2F4PlayRunner()
        {
            EditorApplication.update += PollRequest;
        }

        private static void PollRequest()
        {
            if (running || EditorApplication.isCompiling ||
                EditorApplication.isUpdating || !File.Exists(RequestPath))
            {
                return;
            }
            if (!EditorApplication.isPlaying)
            {
                if (!EditorApplication.isPlayingOrWillChangePlaymode)
                    EditorApplication.EnterPlaymode();
                return;
            }

            running = true;
            File.Delete(RequestPath);
            if (File.Exists(ResultPath))
                File.Delete(ResultPath);
            try
            {
                var tests =
                    new NTSD28B5OrdinaryCreditGate2F4CorrectionEditorTests();
                tests.StandardDamage_LegacyStatsEligibilityReadsExactGate();
                tests.ReducedAndCpointDamage_LegacyStatsEligibilityReadsExactGate();
                tests.CharacterRecovery_DataAndLegacyThresholdReadExactGate();
                tests.LogicOpoint_PropagatesExactGateOnlyForType0Children();
                tests.ProductionSources_ContainNoScopedKillCountBinding();
                File.WriteAllText(
                    ResultPath,
                    "state=Passed\ncases=5\n" +
                    "damageGates=standard,reduced,cpoint\n" +
                    "recovery=data,legacy\nopoint=type0,nonType0,twoHop\n" +
                    "sceneMutation=none\n",
                    new UTF8Encoding(false));
            }
            catch (Exception exception)
            {
                File.WriteAllText(
                    ResultPath,
                    "state=Failed\n" + exception,
                    new UTF8Encoding(false));
            }
            finally
            {
                EditorApplication.delayCall += ExitPlayMode;
            }
        }

        private static void ExitPlayMode()
        {
            if (EditorApplication.isPlaying)
                EditorApplication.ExitPlaymode();
            running = false;
        }

        private static string ProjectPath(string relativePath)
        {
            string root = Directory.GetParent(Application.dataPath)?.FullName;
            return Path.GetFullPath(Path.Combine(root ?? string.Empty, relativePath));
        }
    }
