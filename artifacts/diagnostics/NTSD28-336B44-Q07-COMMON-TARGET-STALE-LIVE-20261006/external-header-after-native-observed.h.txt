// Native knockout bookkeeping emitted immediately before a lethal HP
// subtraction.  FUN_0045C2F0/FUN_0045C320 increment Entity28+0x358 on the
// credited entity and forward the victim/source slot pair to FUN_0045C070.
struct WorldKnockoutEvent28 {
    std::uint64_t sequence = 0;
    // FUN_0045C070 writes a 0x20-byte native result cell.  The first five
    // dwords are battle time, source definition type, the source followed
    // through four +0x354 owners, source slot, and victim slot.
    std::uint64_t battle_time_ticks = 0;
    int source_object_type = -1;
    std::size_t four_owner_slot = EngineProfile28::maximum_slots;
    std::size_t victim_slot = EngineProfile28::maximum_slots;
    std::size_t source_slot = EngineProfile28::maximum_slots;
    // Wrapper-specific +0x358 owner.  This is intentionally separate from
    // four_owner_slot: ordinary damage resolves two owners before entering
    // FUN_0045C3A0, while FUN_0045C070 independently follows four for its
    // persistent result record.
    std::size_t credit_slot = EngineProfile28::maximum_slots;
};
