def utc():
    return datetime.now(timezone.utc).isoformat()


def digest(data):
    return hashlib.sha256(data).hexdigest()


def new_bytes(path, data):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("xb") as stream:
        stream.write(data)


def new_json(path, value):
    new_bytes(path, (json.dumps(value, ensure_ascii=True, indent=2) + "\n").encode("ascii"))


def new_png(path, image):
    data = io.BytesIO()
    image.save(data, format="PNG")
    new_bytes(path, data.getvalue())


def metadata(path):
    data = path.read_bytes()
    return {"path": str(path.relative_to(ROOT)).replace("\\", "/"),
            "bytes": len(data), "sha256": digest(data)}


def assert_inputs(selection):
    for item in selection["samples"]:
        for key in ("source", "decoded_dat"):
            entry = item[key]
            assert metadata(ROOT / entry["path"]) == entry, (key, entry["path"])
    assert metadata(CONTENT / "catalog.csv") == selection["catalog"]


def prepare():
    assert not (OUT / "selection.json").exists(), "Frozen selection already exists"
    catalog = list(csv.DictReader((CONTENT / "catalog.csv").open(encoding="utf-8-sig")))
    result = {"utc": utc(), "profile": "LDR sRGB (-cs/-ds), straight RGBA8",
              "quality": "thorough", "alpha_weight_radius": 1, "threads": 2,
              "mipmaps": False, "resize": False, "atlas_or_paging": False,
              "project_color_space": "Gamma (m_ActiveColorSpace: 0)",
              "catalog": metadata(CONTENT / "catalog.csv"), "samples": []}
    pattern = re.compile(r"file\((\d+)-(\d+)\):\s*(\S+)\s+w:\s*(\d+)\s+h:\s*(\d+)\s+row:\s*(\d+)\s+col:\s*(\d+)")
    for slug, label, rel, dat in SAMPLES:
        src = CONTENT / "vfs" / rel
        dec = CONTENT / "decoded_dat" / dat
        matches = [m for line in dec.read_text(encoding="utf-8-sig").splitlines()
                   if (m := pattern.search(line)) and m[3].replace("\\", "/").lower() == rel.lower()]
        assert len(matches) == 1, (rel, "exact DAT file declaration required")
        m = matches[0]
        start, end, width, height, columns, rows = map(int, (m[1], m[2], m[4], m[5], m[6], m[7]))
        with Image.open(src) as image:
            original_mode = image.mode
            rgba = image.convert("RGBA")
        array = np.array(rgba)
        assert rgba.width * rgba.height <= 2_000_000, "Bounded representative sample only"
        mask = np.zeros(array.shape[:2], dtype=bool)
        rects = []
        for index in range(end - start + 1):
            x = (index % columns) * (width + 1)
            y = (index // columns) * (height + 1)
            assert index // columns < rows
            x1, y1 = min(x + width, rgba.width), min(y + height, rgba.height)
            if x1 <= x or y1 <= y:
                continue
            mask[y:y1, x:x1] = True
            rects.append([x, y, x1, y1])
        alpha = array[:, :, 3]
        foreground = mask & (alpha > 0)
        semi = mask & (alpha > 0) & (alpha < 255)
        target = OUT / "inputs" / (slug + "-rgba.png")
        if not target.exists():
            new_png(target, rgba)
        with Image.open(target) as check:
            assert np.array_equal(np.array(check.convert("RGBA")), array)
        mask_path = OUT / "inputs" / (slug + "-mask.png")
        if not mask_path.exists():
            new_png(mask_path, Image.fromarray(mask.astype(np.uint8) * 255))
        with Image.open(mask_path) as check:
            assert np.array_equal(np.array(check), mask.astype(np.uint8) * 255)
        result["samples"].append({"slug": slug, "label": label, "logical_path": rel,
            "source": metadata(src), "decoded_dat": metadata(dec),
            "catalog_entries": [{k: row[k] for k in ("id", "type", "name", "source_path")}
                                for row in catalog if row["source_path"].replace("\\", "/") == dat],
            "dat_declaration": m[0], "width": rgba.width, "height": rgba.height,
            "original_png_mode": original_mode, "rgba_bytes": rgba.width * rgba.height * 4,
            "rgba_input": metadata(target), "rects": rects,
            "valid_pixels": int(mask.sum()), "visible_pixels": int(foreground.sum()),
            "semi_pixels": int(semi.sum()), "alpha_levels_valid": len(np.unique(alpha[mask]))})
    assert_inputs(result)
    new_json(OUT / "selection.json", result)
    print(json.dumps({"stage": "prepare", "samples": len(result["samples"]),
                      "rgba_bytes": sum(x["rgba_bytes"] for x in result["samples"])}), flush=True)


def command(args, log):
    started = utc()
    before = time.monotonic()
    result = subprocess.run(args, capture_output=True, timeout=240)
    new_bytes(OUT / "logs" / (log + ".txt"), result.stdout + result.stderr)
    receipt = {"command": args, "start_utc": started, "end_utc": utc(),
               "wall_seconds": time.monotonic() - before, "exit_code": result.returncode}
    new_json(OUT / "logs" / (log + ".json"), receipt)
    assert result.returncode == 0, (log, result.returncode)
    return receipt


def astc_info(path, width, height, block):
    data = path.read_bytes()
    assert data[:4] == bytes.fromhex("13aba15c"), "ASTC magic"
    assert list(data[4:7]) == [block, block, 1], "ASTC block dimensions"
    dims = [int.from_bytes(data[i:i + 3], "little") for i in (7, 10, 13)]
    assert dims == [width, height, 1], (path, dims)
    payload = math.ceil(width / block) * math.ceil(height / block) * 16
    assert len(data) == 16 + payload, (path, len(data), payload)
    return {"block": block, "width": width, "height": height, "mip_levels": 1,
            "blocks": payload // 16, "payload_bytes": payload, "header_bytes": 16,
            "file_bytes": len(data), "sha256": digest(data),
            "ideal_gpu_texel_bytes": payload, "device_allocation_alignment": "NOT_MEASURED"}


def encode():
    selection = json.loads((OUT / "selection.json").read_text(encoding="ascii"))
    assert_inputs(selection)
    tool = json.loads((OUT / "encoder/tool-identity.json").read_text(encoding="ascii"))
    exe = tool["selected_executable"]
    expected = next(x["sha256"] for x in tool["files"] if x["path"].endswith("astcenc-sse2.exe"))
    assert digest(Path(exe).read_bytes()) == expected
    results = []
    for sample in selection["samples"]:
        for block in BLOCKS:
            slug = sample["slug"]
            source = OUT / "inputs" / (slug + "-rgba.png")
            assert metadata(source) == sample["rgba_input"]
            compressed = OUT / "compressed" / f"{slug}-{block}x{block}.astc"
            decoded = OUT / "decoded" / f"{slug}-{block}x{block}.png"
            assert not compressed.exists() and not decoded.exists(), "No overwrite/re-encode"
            compressed.parent.mkdir(parents=True, exist_ok=True)
            decoded.parent.mkdir(parents=True, exist_ok=True)
            enc = command([exe, "-cs", str(source), str(compressed), f"{block}x{block}", *OPTIONS], f"{slug}-{block}x{block}-encode")
            info = astc_info(compressed, sample["width"], sample["height"], block)
            dec = command([exe, "-ds", str(compressed), str(decoded), "-j", "2"], f"{slug}-{block}x{block}-decode")
            with Image.open(decoded) as image:
                assert image.size == (sample["width"], sample["height"])
            results.append({"slug": slug, "block": block, "astc": info,
                            "decoded": metadata(decoded), "encode": enc, "decode": dec})
            print(json.dumps({"slug": slug, "block": block, "payload_bytes": info["payload_bytes"],
                              "encode_seconds": enc["wall_seconds"]}), flush=True)
    assert_inputs(selection)
    new_json(OUT / "encode-results.json", {"utc": utc(), "results": results})


def dilate(mask):
    padded = np.pad(mask, 1)
    return np.logical_or.reduce([padded[y:y + mask.shape[0], x:x + mask.shape[1]]
                                 for y in range(3) for x in range(3)])


def measure(delta, mask):
    values = np.asarray(delta, dtype=np.float64)[mask]
    if not len(values):
        return None
    error = np.abs(values)
    mse = float(np.mean(values * values))
    return {"pixels": int(mask.sum()), "mae_8bit": float(error.mean()),
            "p99_abs_8bit": float(np.percentile(error, 99)), "max_abs_8bit": float(error.max()),
            "psnr_db": 99.0 if mse == 0 else 10 * math.log10(255 * 255 / mse)}


def composites(array, background):
    alpha = array[:, :, 3:4].astype(np.float64) / 255
    return array[:, :, :3] * alpha + background * (1 - alpha)


def rgba(path):
    with Image.open(path) as image:
        return np.array(image.convert("RGBA"))


def create_board(rows):
    board = Image.new("RGB", (1200, 88 + len(rows) * 288), "#111827")
    draw = ImageDraw.Draw(board)
    font = ImageFont.truetype("C:/Windows/Fonts/segoeui.ttf", 18)
    small = ImageFont.truetype("C:/Windows/Fonts/segoeui.ttf", 14)
    draw.text((20, 12), "ASTC sample comparison | nearest-neighbor zoom | identical DAT frame", font=font, fill="white")
    for col, title in enumerate(("Original RGBA", "ASTC 4x4", "ASTC 6x6")):
        draw.text((20 + col * 400, 48), title, font=font, fill="#93c5fd")
    for row, sample in enumerate(rows):
        y0 = 88 + row * 288
        for col, variant in enumerate(("original", "4x4", "6x6")):
            image = Image.open(OUT / sample["preview_files"][variant]).convert("RGBA")
            scale = max(1, min(4, 370 // image.width, 230 // image.height))
            image = image.resize((image.width * scale, image.height * scale), Image.Resampling.NEAREST)
            yy, xx = np.indices((image.height, image.width))
            checker = np.where((((xx // 12) + (yy // 12)) % 2)[..., None], 70, 120)
            checker = Image.fromarray(np.repeat(checker.astype(np.uint8), 3, axis=2), "RGB")
            checker.paste(image, (0, 0), image)
            x0 = 20 + col * 400
            board.paste(checker, (x0, y0 + 42))
            label = sample["label"] if col == 0 else f"alpha MAE {sample['metrics'][variant]['alpha_visible']['mae_8bit']:.2f} / 255"
            draw.text((x0, y0 + 5), label, font=small, fill="white")
            draw.text((x0, y0 + 258), f"frame {sample['preview_frame']} | zoom {scale}x", font=small, fill="#9ca3af")
    new_png(OUT / "comparison-overview.png", board)


def report():

    encoded = json.loads((OUT / "encode-results.json").read_text(encoding="ascii"))
    assert_inputs(selection)
    rows = []
    for sample in selection["samples"]:
        slug = sample["slug"]
        original = rgba(OUT / "inputs" / (slug + "-rgba.png"))
        valid = rgba(OUT / "inputs" / (slug + "-mask.png"))[:, :, 0] > 0
        alpha = original[:, :, 3]
        visible = valid & (alpha > 0)
        semi = visible & (alpha < 255)
        padded = np.pad(alpha.astype(np.int16), 1, mode="edge")
        neighbors = [padded[y:y + alpha.shape[0], x:x + alpha.shape[1]] for y in range(3) for x in range(3)]
        edge = valid & ((np.maximum.reduce(neighbors) != np.minimum.reduce(neighbors)) | semi)
        near = valid & dilate(visible)
        variants = {"original": original}
        metrics = {}
        byte_results = {}
        for block in BLOCKS:
            name = f"{block}x{block}"
            decoded = rgba(OUT / "decoded" / f"{slug}-{name}.png")
            variants[name] = decoded
            delta = decoded.astype(np.int16) - original.astype(np.int16)
            metric = {"rgb_visible": measure(delta[:, :, :3], visible),
                      "alpha_valid": measure(delta[:, :, 3], valid),
                      "alpha_visible": measure(delta[:, :, 3], visible),
                      "alpha_semi": measure(delta[:, :, 3], semi),
                      "alpha_edge": measure(delta[:, :, 3], edge),
                      "opaque_alpha_under_250": int((valid & (alpha == 255) & (decoded[:, :, 3] < 250)).sum()),
                      "transparent_alpha_over_5": int((valid & (alpha == 0) & (decoded[:, :, 3] > 5)).sum()),
                      "composite_srgb_visible_and_neighbor": {}}
            yy, xx = np.indices(alpha.shape)
            for bgname, bg in [("black", np.array([0, 0, 0])), ("white", np.array([255, 255, 255])),
                               ("gray", np.array([128, 128, 128])),
                               ("checker", np.repeat(np.where(((xx // 8 + yy // 8) % 2)[..., None], 64, 192), 3, axis=2))]:
                metric["composite_srgb_visible_and_neighbor"][bgname] = measure(composites(decoded, bg) - composites(original, bg), near)
            metrics[name] = metric
            byte_results[name] = next(x["astc"] for x in encoded["results"] if x["slug"] == slug and x["block"] == block)
        # Freeze a content-rich frame using only the original, never choose by a codec's score.
        def score(rect):
            x0, y0, x1, y1 = rect
            a = alpha[y0:y1, x0:x1]
            return int(((a > 0) & (a < 255)).sum()) if sample["slug"] in ("smoke", "flame") else int((a > 0).sum())
        frame_index = max(range(len(sample["rects"])), key=lambda index: score(sample["rects"][index]))
        rect = sample["rects"][frame_index]
        # Large flame frame also gets a fixed center ROI for exact nearest zoom, without changing encoded sheet.
        x0, y0, x1, y1 = rect
        if x1 - x0 > 160 or y1 - y0 > 160:
            cx, cy = (x0 + x1) // 2, (y0 + y1) // 2
            rect = [cx - 72, cy - 72, cx + 72, cy + 72]
        previews = {}
        for name, array in variants.items():
            target = OUT / "previews" / f"{slug}-{name}-roi.png"
            new_png(target, Image.fromarray(array).crop(tuple(rect)))
            previews[name] = str(target.relative_to(OUT)).replace("\\", "/")
        rows.append({**sample, "metrics": metrics, "astc_bytes": byte_results,
                     "preview_frame": frame_index, "preview_rect_top_left": rect, "preview_files": previews})
    totals = {"rgba_bytes": sum(x["rgba_bytes"] for x in rows),
              "source_png_file_bytes": sum(x["source"]["bytes"] for x in rows)}
    for block in BLOCKS:
        name = f"{block}x{block}"
        totals[name] = {"payload_bytes": sum(x["astc_bytes"][name]["payload_bytes"] for x in rows),
                        "file_bytes": sum(x["astc_bytes"][name]["file_bytes"] for x in rows)}
    new_json(OUT / "metrics.json", {"utc": utc(), "method": "Full sheets encoded. DAT cells exclude separators from metrics; original-only foreground/alpha/edge masks; straight alpha sRGB compositing (project Gamma). No mipmaps/resize/premultiplication. Software decoded PNGs are visual evidence, not GPU measurements.", "totals": totals, "samples": rows})
    create_board(rows)
    ui = []
    for sample in rows:
        sources = {}
        for variant in ("original", "4x4", "6x6"):
            path = OUT / "inputs" / (sample["slug"] + "-rgba.png") if variant == "original" else OUT / "decoded" / f"{sample['slug']}-{variant}.png"
            sources[variant] = "data:image/png;base64," + base64.b64encode(path.read_bytes()).decode("ascii")
