11:         {
12:             if (block == null) throw new ArgumentNullException(nameof(block));
13:             var armor = new LF2ArmorData();
14:             int ptype = 0;

22:                     case "type": validType = LoganNumericDecoder.TryParseInt32(field.Value, out armor.type); break;

24:                     case "ratio": armor.ratio = value; break;
25:                     case "decrease": armor.decrease = value; break;
26:                     case "mp": armor.mp = value; break;
27:                     case "fall": armor.fall = value; break;
28:                     case "bdefend": armor.bdefend = value; break;
29:                     case "injury": armor.injury = value; break;
30:                     case "spark": armor.spark = value; break;
31:                     case "hp": armor.hp = value; break;
32:                     case "recover": armor.recover = value; break;
33:                     case "facing": armor.facing = value; break;
34:                     case "action": armor.action = value; break;
35:                     case "reserve": armor.reserve = value; break;
36:                     case "delay": armor.delay = value; break;
37:                     case "sound1": armor.sound1 = field.Value; break;
38:                     case "sound2": armor.sound2 = field.Value; break;
39:                     case "frame":
40:                         if (TryReadArmorInteger(field.Value, ref cursor, out int first) &&
41:                             TryReadArmorInteger(field.Value, ref cursor, out int last))
42:                             armor.frame_ranges.Add(new LF2ArmorFrameRange { first = first, last = last });
43:                         break;
44:                     case "state": if (TryReadArmorInteger(field.Value, ref cursor, out value)) armor.states.Add(value); break;
45:                     case "kind": if (TryReadArmorInteger(field.Value, ref cursor, out value)) armor.kinds.Add(value); break;
46:                     case "id": if (TryReadArmorInteger(field.Value, ref cursor, out value)) armor.ids.Add(value); break;
47:                     case "effect": if (TryReadArmorInteger(field.Value, ref cursor, out value)) armor.effects.Add(value); break;
48:                 }
49:             }
50:             if (!validType) armor.type = ptype;
51:             return armor;
52:         }
