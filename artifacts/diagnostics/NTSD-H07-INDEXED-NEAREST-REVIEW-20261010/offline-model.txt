def i32(x): return ((x+2**31)%2**32)-2**31
def ab(x): return i32(-x) if x<0 else x
def sub(a,b): return i32(a-b)
def team_allowed(a,b,p):
    if b!=a:
        if p!=1 or a==5: return True
    if b!=5 or p!=1: return False
    return b!=a
def included(r,s): return 0<=s<len(r) and r[s]['included']
def role(r,s,air):
    q=r[s]
    if not q['included'] or q['hp']<=0: return False
    return (q['state']==14 or ab(q['stop'])>2) if air else (q['state']!=14 and ab(q['stop'])<=2 and (q['type']==0 or q['state']==3000))
def target(r,s,t,p,air):
    if s==t or not included(r,t): return False
    a,b=r[s],r[t]
    if air: return team_allowed(a['team'],b['team'],p) and b['hp']>0 and (b['state']==14 or ab(b['stop'])>2)
    if b['type']!=0:
        if b['state']!=3000: return False
        if b['x']>a['x']:
            if not b['vx']<.001: return False
        elif b['x']<a['x']:
            if not b['vx']>.001: return False
        else: return False
    return team_allowed(a['team'],b['team'],p) and b['hp']>0 and b['state']!=14 and ab(b['stop'])<=2

def search(r,s,p,air,reuse):
    groups={}; visits=[]; selected=-1; best=10000; evaluations=0
    for slot in range(len(r)):
        if role(r,slot,air): groups.setdefault(r[slot]['team'],[]).append(slot)
    for tm,slots in sorted(groups.items()):
        slots.sort(key=lambda q:(r[q]['x'],q))
        if not team_allowed(r[s]['team'],tm,p): continue
        right=bisect.bisect_left([r[q]['x'] for q in slots],r[s]['x']); left=right-1
        lc=rc=None; ld=rd=2**31-1
        while left>=0 or right<len(slots):
            if not reuse or lc!=left:
                ld=ab(sub(r[slots[left]]['x'],r[s]['x'])) if left>=0 else 2**31-1
                evaluations+=int(left>=0); lc=left
            if not reuse or rc!=right:
                rd=ab(sub(r[slots[right]]['x'],r[s]['x'])) if right<len(slots) else 2**31-1
                evaluations+=int(right<len(slots)); rc=right
            limit=min(best,249) if air else best
            if ld>limit and rd>limit: break
            if ld<=rd: slot=slots[left]; left-=1; dx=ld
            else: slot=slots[right]; right+=1; dx=rd
            visits.append(slot)
            if not target(r,s,slot,p,air): continue
            dz=ab(sub(r[slot]['z'],r[s]['z']))
            if not reuse: dx=ab(sub(r[slot]['x'],r[s]['x'])); evaluations+=1
            dist=i32(dx+dz)
            if air and (dz>=40 or dx>=250): continue
            if dist<best or (dist==best and selected>=0 and slot<selected): best,selected=dist,slot
    return selected,best,visits,evaluations

def nearest(r,s,p,reuse,ready=True):
    if not ready or not included(r,s) or r[s]['coord']>-1000: return None,0
    selected,best,gv,gc=search(r,s,p,False,reuse)
    lane=selected>=0 and ab(sub(r[selected]['z'],r[s]['z']))<15
    av=[]; ac=0
    if r[s]['state']!=9:
        air,_,av,ac=search(r,s,p,True,reuse)
        if air>=0: selected=air
    generation=r[selected]['generation'] if selected>=0 else 0
    identity=r[selected]['identity'] if selected>=0 else 0
    return (selected,best,lane,91,generation,identity,len(gv),len(av),tuple(gv),tuple(av)),gc+ac

def linear(r,s,p):
    if not included(r,s) or r[s]['coord']>-1000: return None
    sel=-1; best=10000
    for q in range(len(r)):
        if target(r,s,q,p,False):
            d=i32(ab(sub(r[q]['x'],r[s]['x']))+ab(sub(r[q]['z'],r[s]['z'])))
            if d<best: best,sel=d,q
    lane=sel>=0 and ab(sub(r[sel]['z'],r[s]['z']))<15
    if r[s]['state']!=9:
        air=-1; b=10000
        for q in range(len(r)):
            if not target(r,s,q,p,True): continue
            dz=ab(sub(r[q]['z'],r[s]['z'])); dx=ab(sub(r[q]['x'],r[s]['x'])); d=i32(dx+dz)
            if d>=b or dz>=40 or dx>=250: continue
            b,air=d,q
        if air>=0: sel=air
    return sel,best,lane,91,r[sel]['generation'] if sel>=0 else 0,r[sel]['identity'] if sel>=0 else 0
