def transform(raw,edit):
    bom=b"\xef\xbb\xbf" if raw.startswith(b"\xef\xbb\xbf") else b""
    body=raw[len(bom):].decode("utf-8")
    nl="\r\n" if "\r\n" in body else "\n"
    for old,new in edit.get("replace",[]):
        old=old.replace("\n",nl);new=new.replace("\n",nl)
        assert body.count(old)==1,(edit["path"],body.count(old),old)
        body=body.replace(old,new,1)
    for table in [edit.get("reorder"),edit.get("childReorder")]:
        if not table:continue
        pattern=r"(^"+re.escape(table["header"])+r"\r?\n\|[^\n]+\r?\n)((?:\|[^\n]*\r?\n)+)"
        matches=list(re.finditer(pattern,body,re.M))
        assert len(matches)==1,(edit["path"],table["header"],len(matches))
        match=matches[0]
        rows=match.group(2).splitlines()
        keyed={}
        for row in rows:
            token=re.match(r"^\|\s*(?:\[)?([A-Z][A-Za-z0-9]*(?:-[A-Za-z0-9]+)+)(?:\]|\s|\|)",row)
            assert token,(edit["path"],row)
            assert token.group(1) not in keyed
            keyed[token.group(1)]=row
        assert set(keyed)==set(table["ids"]),(edit["path"],list(keyed),table["ids"])
        replacement=match.group(1)+nl.join(keyed[key] for key in table["ids"])+nl
        body=body[:match.start()]+replacement+body[match.end():]
    return bom+(edit["prepend"].replace("\n",nl)+body).encode("utf-8")
