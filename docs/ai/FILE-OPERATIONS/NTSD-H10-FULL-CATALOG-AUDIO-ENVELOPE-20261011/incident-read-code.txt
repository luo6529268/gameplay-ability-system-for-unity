var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic;
var cache = (System.Collections.IDictionary)typeof(NTSD.Load.NTSD_ResourceLoader).GetField("cache", flags).GetValue(NTSD.Load.NTSD_ResourceLoader.Instance);
var roots = UnityEngine.SceneManagement.SceneManager.GetActiveScene().GetRootGameObjects().Select(x => new { id=x.GetInstanceID(), x.name, types=x.GetComponents<UnityEngine.Component>().Where(c=>c!=null).Select(c=>c.GetType().FullName).ToArray() }).ToArray();
var values = cache.Keys.Cast<string>().Select(k => new { key=k, type=cache[k].GetType().FullName, clipId=cache[k] is UnityEngine.AudioClip ? ((UnityEngine.AudioClip)cache[k]).GetInstanceID() : 0 }).ToArray();
return new { roots, values, nativeClips=UnityEngine.Resources.FindObjectsOfTypeAll<UnityEngine.AudioClip>().Select(x=>new { id=x.GetInstanceID(), x.name, x.samples, x.channels, x.frequency }).ToArray(), pending=NTSD.Load.NTSD_ResourceLoader.Instance.GetPendingRequestCount(), queued=NTSD.Load.NTSD_ResourceLoader.Instance.GetQueuedTaskCount(), soundManagerCurrent=MoreMountains.Tools.MMSoundManager.Current!=null?MoreMountains.Tools.MMSoundManager.Current.GetInstanceID():0 };
