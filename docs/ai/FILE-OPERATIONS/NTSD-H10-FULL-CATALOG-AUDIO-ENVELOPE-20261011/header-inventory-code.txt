from pathlib import Path
import json,struct,hashlib,datetime,collections
out=Path('artifacts/diagnostics/NTSD-H10-FULL-CATALOG-AUDIO-ENVELOPE-20261011')
inventory=json.loads((out/'inventory.json').read_text(encoding='utf-8'))
def header(path):
 p=Path(path);r={'path':path,'fileBytes':p.stat().st_size,'knownFrames':False}
 with p.open('rb') as f:
  start=f.read(12)
  if start[:4]!=b'RIFF' or start[8:]!=b'WAVE':r['reason']='non-RIFF/WAVE';return r
  fmt=None;fact=None;dataSize=0
  while f.tell()+8<=p.stat().st_size:
   raw=f.read(8);kind,size=struct.unpack('<4sI',raw);pos=f.tell()
   if kind==b'fmt ':fmt=f.read(min(size,40))
   elif kind==b'fact' and size>=4:fact=struct.unpack('<I',f.read(4))[0]
   elif kind==b'data':dataSize+=size
   f.seek(pos+size+(size%2))
  if fmt is None:r['reason']='missing fmt';return r
  form,ch,hz,rate,align,bits=struct.unpack('<HHIIHH',fmt[:16])
  frames=dataSize//align if form in (1,3) and align else fact
  r.update({'format':form,'channels':ch,'frequency':hz,'blockAlign':align,'bitsPerSample':bits,'dataBytes':dataSize,'frames':frames,'knownFrames':frames is not None,'reason':None if frames is not None else 'compressed without fact'})
  if frames is not None:r['sourceFloatPcmBytes']=frames*ch*4;r['battleFloatPcmBytes']=frames*(2 if ch==1 else ch)*4
 return r
headers={x['path']:header(x['path']) for x in inventory['files']}
formal={x['battle']['path'] for x in inventory['routes'] if x['battle']['formal'] and x['battle']['exists']}
generic={p for x in inventory['routes'] if x['generic']['exists'] for p in x['generic']['files']}
summary={'formalUniqueFiles':len(formal),'genericUniqueFiles':len(generic),'genericExistingRoutes':inventory['genericExistingRoutes'],'sourcePcmBytes':sum(headers[p].get('sourceFloatPcmBytes',0) for p in formal),'formalPlaybackPcmBytes':sum(headers[p].get('battleFloatPcmBytes',0) for p in formal),'genericPcmBytes':sum(headers[p].get('sourceFloatPcmBytes',0) for p in generic),'unknownHeaders':[p for p,h in headers.items() if not h['knownFrames']],'formats':dict(collections.Counter(str(h.get('format')) for h in headers.values()))}
result={'inventorySha256':hashlib.sha256((out/'inventory.json').read_bytes()).hexdigest(),'files':headers,'summary':summary,'scope':'Header PCM payload only, not native object allocation, decoder peak, RSS or approved budget'}
with (out/'headers.json').open('x',encoding='utf-8') as f:json.dump(result,f,ensure_ascii=False,indent=2)
print(json.dumps(summary))
