function Read-JsonGzip([string]$path){
 $inputStream=[IO.File]::OpenRead($path)
 $gzip=[IO.Compression.GZipStream]::new($inputStream,[IO.Compression.CompressionMode]::Decompress)
 $reader=[IO.StreamReader]::new($gzip,[Text.UTF8Encoding]::new($false))
 try{$content=$reader.ReadToEnd()}finally{$reader.Dispose();$gzip.Dispose();$inputStream.Dispose()}
 return $content|ConvertFrom-Json
}

Add-Type -TypeDefinition @'
using System;
using System.IO;
using System.Security.Cryptography;
using System.Threading.Tasks;
public sealed class DiagnosticHashObservation
{
    public string Path { get; set; }
    public long Bytes { get; set; }
    public string Sha256 { get; set; }
    public string Error { get; set; }
}
public static class DiagnosticReadOnlyHashVerifier
{
    public static DiagnosticHashObservation[] Read(string root, string[] paths)
    {
        var observations = new DiagnosticHashObservation[paths.Length];
        var prefix = System.IO.Path.GetFullPath(root) + System.IO.Path.DirectorySeparatorChar;
        Parallel.For(0, paths.Length, new ParallelOptions { MaxDegreeOfParallelism = 4 }, i =>
        {
            var value = new DiagnosticHashObservation { Path = paths[i] };
            observations[i] = value;
            try
            {
                var absolute = System.IO.Path.GetFullPath(System.IO.Path.Combine(root, paths[i]));
                if (!absolute.StartsWith(prefix, StringComparison.OrdinalIgnoreCase))
                    throw new InvalidOperationException("Read-only path escaped repository");
                var before = new FileInfo(absolute);
                var length = before.Length;
                var ticks = before.LastWriteTimeUtc.Ticks;
                using (var stream = new FileStream(absolute, FileMode.Open, FileAccess.Read, FileShare.Read))
                {
                    value.Bytes = stream.Length;
                    value.Sha256 = Convert.ToHexString(SHA256.HashData(stream)).ToLowerInvariant();
                }
                var after = new FileInfo(absolute);
                if (after.Length != length || after.LastWriteTimeUtc.Ticks != ticks)
                    throw new IOException("File changed during read-only hash verification");
            }
            catch (Exception error) { value.Error = error.ToString(); }
        });
        return observations;
    }
}

function Patch-Map([string]$path){
 $map=[Collections.Generic.Dictionary[string,string]]::new([StringComparer]::Ordinal)
 $text=[IO.File]::ReadAllText($path,[Text.UTF8Encoding]::new($false)).Replace(([string][char]13+[string][char]10),([string][char]10))
 foreach($block in [regex]::Split($text,'(?m)(?=^diff --git )')){
  if(-not $block.StartsWith('diff --git ')){continue}
  $header=($block -split ([string][char]10),2)[0]
  if($allowedHeaders -contains $header){continue}
  $map.Add($header,[Convert]::ToHexString([Security.Cryptography.SHA256]::HashData([Text.UTF8Encoding]::new($false).GetBytes($block))).ToLowerInvariant())
 }
 return ,$map
}
